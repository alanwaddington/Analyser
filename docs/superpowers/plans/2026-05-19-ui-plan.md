## Overview

Build the full client-side UI for the Analyser app on top of the existing SvelteKit + TypeScript + TailwindCSS scaffold. ### Step 3: Create the session store

**What:** Create `src/lib/stores/session.ts` with all state fields specified in the spec.

**File:** `src/lib/stores/session.ts` (create)

**Change:**
```typescript
import { writable, derived } from 'svelte/store';
import type { Activity, ChannelKey } from '$lib/types';

export const activities = writable<Activity[]>([]);
export const smoothing = writable<number>(10);
export const xAxisMode = writable<'time' | 'distance'>('time');
export const referenceIndex = writable<number>(0);
export const clearing = writable<boolean>(false);
export const lastMode = writable<'compare' | 'event'>('compare');
export const activeChannels = writable<ChannelKey[]>([]);

export function addActivity(activity: Activity): void {
  activities.update(list => [...list, activity]);
}

export function removeActivity(id: string): void {
  referenceIndex.update(ref => {
    let newRef = ref;
    activities.update(list => {
      const idx = list.findIndex(a => a.id === id);
      if (idx === -1) return list;
      if (idx < ref) newRef = ref - 1;
      if (idx === ref) newRef = 0;
      return list.filter(a => a.id !== id);
    });
    return newRef;
  });
}

export function clearActivities(): void {
  activities.set([]);
  referenceIndex.set(0);
  activeChannels.set([]);
}
```

### Step 4: Create `DropZone.svelte`

**What:** Reusable file drop zone. `compact=false` renders a full centred empty-state panel; `compact=true` renders a small dashed "Add files" button. Both handle drag-drop and click-to-open. Calls `addActivity` from the store after parsing each file via the existing `parseFile` function from `$lib/fit/parser`.

**File:** `src/lib/components/ui/DropZone.svelte` (create)

**Change:**
```svelte
<script lang="ts">
  import { parseFile } from '$lib/fit/parser';
  import { activities, addActivity } from '$lib/stores/session';
  import { MAX_FILES } from '$lib/types';
  import { get } from 'svelte/store';

  let { compact = false }: { compact?: boolean } = $props();

  let dragover = $state(false);
  let warning = $state('');

  async function handleFiles(files: FileList | File[]) {
    const arr = Array.from(files).filter(f => f.name.endsWith('.fit'));
    const current = get(activities);
    const slots = MAX_FILES - current.length;
    if (slots <= 0) {
      warning = 'Maximum 6 files — remove one before adding another';
      return;
    }
    const toLoad = arr.slice(0, slots);
    if (arr.length > slots) {
      warning = 'Maximum 6 files — some files were not loaded';
    } else {
      warning = '';
    }
    for (const file of toLoad) {
      try {
        const activity = await parseFile(file);
        addActivity(activity);
      } catch (e) {
        console.error('Failed to parse', file.name, e);
      }
    }
  }

  function onDrop(e: DragEvent) {
    e.preventDefault();
    dragover = false;
    if (e.dataTransfer?.files) handleFiles(e.dataTransfer.files);
  }

  function onDragOver(e: DragEvent) {
    e.preventDefault();
    dragover = true;
  }

  function onDragLeave() {
    dragover = false;
  }

  function onFileInput(e: Event) {
    const input = e.currentTarget as HTMLInputElement;
    if (input.files) handleFiles(input.files);
    input.value = '';
  }
</script>

{#if compact}
  <label
    class="dropzone-compact"
    class:dragover
    ondrop={onDrop}
    ondragover={onDragOver}
    ondragleave={onDragLeave}
  >
    <input type="file" multiple accept=".fit" onchange={onFileInput} hidden />
    + Add files
  </label>
  {#if warning}
    <p class="dropzone-warning">{warning}</p>
  {/if}
{:else}
  <label
    class="dropzone-full"
    class:dragover

    <span class="dropzone-browse">Browse files</span>

{/if}
```

```svelte
  function formatTime(seconds: number): string {
    const m = Math.floor(seconds / 60);
```

### Step 8: Create `ChannelToggleBar.svelte`

**What:** Row of pill buttons, one per available channel. Reads/writes `activeChannels` store. Receives `channels: ChannelKey[]` prop from parent.

**File:** `src/lib/components/ui/ChannelToggleBar.svelte` (create)

**Change:**
```svelte
<script lang="ts">
  import { activeChannels } from '$lib/stores/session';
  import { CHANNEL_META, type ChannelKey } from '$lib/types';

  let { channels }: { channels: ChannelKey[] } = $props();

  function toggle(ch: ChannelKey) {
    activeChannels.update(active => {
      if (active.includes(ch)) return active.filter(c => c !== ch);
      return [...active, ch];
    });
  }
</script>
```

### Step 11: Rewrite `+page.svelte` (landing / redirect)

**What:** Replace the placeholder. Shows the full DropZone. On mount: if activities are already populated (browser back path), sets `clearing=true`, clears activities, then clears the `clearing` flag. The redirect `$effect` watches `activities.length >= 1 && !clearing` and navigates to `/compare` or `/event`.

**File:** `src/routes/+page.svelte` (modify)

**Change:**
```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  import { goto } from '$app/navigation';
  import { activities, clearing, lastMode, clearActivities } from '$lib/stores/session';
  import { get } from 'svelte/store';
  import DropZone from '$lib/components/ui/DropZone.svelte';

  onMount(() => {
    if (get(activities).length > 0) {
      clearing.set(true);
      clearActivities();
      clearing.set(false);
```

```svelte
  import { onMount } from 'svelte';
  import * as echarts from 'echarts';
```

```svelte
  interface Segment {
    label: string;
    startDist: number;
    endDist: number;
  }

  let {
    activities,

    segments,

    activities: Activity[];
```

```svelte
        polylines.push(poly);
      }
    });
```

```svelte
    const keys = new Set<ChannelKey>();

      'heartRate', 'power', 'cadence', 'speed', 'altitude',
      'temperature', 'coreTemperature', 'verticalOscillation',
      'groundContactTime', 'strideLength',
    ];
    for (const a of $activities) {
      for (const ch of channelList) {
        if (a.records.some(r => r[ch] !== undefined)) keys.add(ch);
      }
    }
    return [...keys];

      activeChannels.set(avail);

    const useDistance = $xAxisMode === 'distance';
```

```svelte
        startDist: l.startDistance,
        endDist: l.endDistance,

    }
    const totalKm = Math.floor(ref.totalDistance / 1000);
    return Array.from({ length: totalKm }, (_, i) => ({
      label: `${i}–${i + 1}km`,
      startDist: i * 1000,
      endDist: (i + 1) * 1000,
    }));
  });

  const lapMarkers = $derived(() => {
    return segments().map(s => ({
      value: $xAxisMode === 'distance' ? s.startDist / 1000 : 0,
      label: s.label,
    })).filter(m => m.value > 0);
  });

  const seriesInputs = $derived(
    $activities.map((a, i) => ({ activity: a, colourIndex: i }))
  );

  function formatTime(seconds: number): string {
    const m = Math.floor(seconds / 60);
    const s = Math.floor(seconds % 60);
    return `${m}:${s.toString().padStart(2, '0')}`;
  }

  function formatDate(d: Date): string {
    return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
  }

  function avgPace(a: { totalDistance: number; totalElapsedTime: number }): string {
    if (a.totalDistance === 0) return '—';
    const secPerKm = a.totalElapsedTime / (a.totalDistance / 1000);
    const m = Math.floor(secPerKm / 60);
    const s = Math.floor(secPerKm % 60);
    return `${m}:${s.toString().padStart(2, '0')}/km`;
  }

  function avgValue(records: any[], field: string): string {
    const vals = records.map(r => r[field]).filter(v => v !== undefined);
    if (!vals.length) return '—';
    return (vals.reduce((a: number, b: number) => a + b, 0) / vals.length).toFixed(0);
  }

  function maxValue(records: any[], field: string): string {
    const vals = records.map(r => r[field]).filter(v => v !== undefined);
    if (!vals.length) return '—';
    return Math.max(...vals).toFixed(0);
  }
```
